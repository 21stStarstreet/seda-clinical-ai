Integration Guide & Live Code Tabs
How to Integrate into Your Projects
Pulsar Glass has zero external dependencies. The tab bar below is also powered by the Pulsar Glass liquid slider! Slide across to explore HTML/JS, Blazor, or React examples: 
<!-- 1. CSS & JS Dosyalarını Dahil Edin -->
<link rel="stylesheet" href="dist/pulsar-glass.css" />
<script src="dist/pulsar-glass.js"></script>

<!-- 2. Segmented Kontrol HTML İskeleti -->
<div class="pg-track" id="zamanAraligi">
  <div class="pg-slider"></div>
  <button class="pg-btn active" data-value="1h"><span class="pg-label">1 Hafta</span></button>
  <button class="pg-btn" data-value="1a"><span class="pg-label">1 Ay</span></button>
  <button class="pg-btn" data-value="3a"><span class="pg-label">3 Ay</span></button>
</div>
// 3. Sayfa Yüklendiğinde Pulsar Glass'ı Başlatın
const slider = PulsarGlass.create("#zamanAraligi", {
  enableRefraction: true,        // Dinamik prizmatik ışık dağılımı
  enableWaterRefraction: true,   // Metin su altı dalga distorsiyonu
  onChange: (index, value, btn) => {
    console.log(`Aktif segment: ${index} (${value})`);
  }
});

// Programatik geçiş (Opsiyonel):
// slider.selectIndex(1); // '1 Ay' seçeneğine kayar
@* Blazor Server veya WebAssembly Projelerinde Tek Satırda Kullanım *@
<PulsarGlassSegmented TItem="string"
                      Items='@(new[] { "1 Hafta", "1 Ay", "3 Ay" })'
                      @bind-Value="secilenAralik"
                      Size="PulsarGlassSize.Standard"
                      OnChange="OnAralikDegisti" />

@code {
    private string secilenAralik = "1 Hafta";

    private void OnAralikDegisti(string yeniDeger) {
        Console.WriteLine($"Seçilen: {yeniDeger}");
    }
}
import React, { useEffect, useRef } from 'react';
import 'pulsar-glass/dist/pulsar-glass.css';
import { PulsarGlass } from 'pulsar-glass';

export function TimeRangeFilter() {
  const trackRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!trackRef.current) return;
    const glass = PulsarGlass.create(trackRef.current, {
      onChange: (idx, val) => console.log('Seçildi:', val)
    });
    return () => glass.destroy?.();
  }, []);

  return (
    <div className="pg-track" ref={trackRef}>
      <div className="pg-slider" />
      <button className="pg-btn active" data-value="1h"><span className="pg-label">1 Hafta</span></button>
      <button className="pg-btn" data-value="1a"><span className="pg-label">1 Ay</span></button>
      <button className="pg-btn" data-value="3a"><span className="pg-label">3 Ay</span></button>
    </div>
  );
}
